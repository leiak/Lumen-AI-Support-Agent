from langchain_core.tools import tool as langchain_tool
from pydantic import BaseModel, Field


class SearchInternalKbArgs(BaseModel):
    query: str = Field(..., description="自然语言查询。")
    kb_slug: str | None = Field(default=None, description="可选:限定特定 KB(空=搜全租户 KB)。")
    top_k: int = Field(default=5, description="返回前 N 条,封顶 20。")


def make_search_internal_kb_tool(*, rag_service, kb_repository):
    """工厂:闭包注入 rag_service 与 kb_repository,运行时从 ContextVar 拿 tenant_id。"""

    @langchain_tool("search_internal_kb", args_schema=SearchInternalKbArgs)
    async def search_internal_kb(query: str, kb_slug: str | None = None, top_k: int = 5) -> str:
        """查询租户私有知识库。可以指定特定 KB(slug),或搜全租户所有 KB。
返回 top-k 命中的 chunk 摘要 + 来源文章标题 + chunk_id。"""
        ctx = _current_escalation_ids()
        if ctx is None:
            return "Error: no active conversation context."
        tenant_id, conversation_id = ctx

        kb = await kb_repository.find_by_slug(tenant_id, kb_slug) if kb_slug else None
        results = await rag_service.retrieve(
            tenant_id=tenant_id,
            conversation_id=conversation_id,
            query=query.strip()[:500],
            knowledge_base_id=kb.id if kb else None,
            top_k=min(top_k, 20),
        )

        if not results:
            return "No relevant articles found."
        lines = [f"Found {len(results)} result(s):"]
        for r in results:
            lines.append(f"- **{r['article_title']}** (score={r['score']:.2f}, chunk={r['chunk_id']}): {r['content'][:200]}")
        return "\n".join(lines)

    return search_internal_kb
```

# 顶部 imports 加
from langchain_core.messages import ToolMessage
MAX_TOOL_ITERATIONS = 5

# 改 make_llm_node 签名 + 实现
def make_llm_node(*, llm_factory, model, tools: list | None = None, max_iterations: int = MAX_TOOL_ITERATIONS):
    """M1: first-wins (tool_calls[:1]). M2.A (Stage 12):全 loop,直到 LLM 不再调 tool。

    tools 是可选的 LangChain BaseTool 列表。空时等同 M1 行为。
    """
    if tools is None:
        tools = []
    tool_map = {t.name: t for t in tools}

    async def llm_node(state: AgentState) -> AgentState:
        from core.logging import get_logger
        log = get_logger(__name__)
        llm = llm_factory(state["tenant_id"])
        bound = llm.bind_tools(tools) if tools else llm
        messages = list(state["messages"])
        iterations = state.get("tool_iterations", 0)

        for _ in range(max_iterations + 1):
            response = await bound.ainvoke(messages)
            messages.append(response)
            if not getattr(response, "tool_calls", None):
                return {"messages": messages, "tool_iterations": 0}

            iterations += 1
            if iterations > max_iterations:
                log.warning("tool_loop_max_iterations_exceeded", iterations=iterations)
                messages.append(AIMessage(content=FALLBACK_MESSAGE))
                return {"messages": messages, "tool_iterations": iterations}

            for tc in response.tool_calls:
                tool = tool_map.get(tc["name"])
                if tool is None:
                    tool_result = f"Error: unknown tool '{tc['name']}'"
                else:
                    try:
                        tool_result = await tool.ainvoke(tc["args"])
                    except Exception as e:  # noqa: BLE001
                        tool_result = f"Error: {type(e).__name__}: {str(e)[:200]}"
                messages.append(ToolMessage(content=str(tool_result), tool_call_id=tc["id"]))

        return {"messages": messages, "tool_iterations": iterations}

    return llm_node
```

> 注:FALLBACK_MESSAGE 是 nodes.py 已有的常量。如果不存在,加 `FALLBACK_MESSAGE = "I'm sorry, I'm having trouble. Let me connect you to a human agent."`

# apps/api/src/ticket/state_machine.py
from ticket.enums import TicketStatus


class InvalidTransition(Exception):
    """Raised when a status transition violates the state machine."""


_TRANSITIONS: dict[TicketStatus, set[TicketStatus]] = {
    TicketStatus.NEW: {TicketStatus.TRIAGED, TicketStatus.CANCELLED},
    TicketStatus.TRIAGED: {TicketStatus.IN_PROGRESS, TicketStatus.CANCELLED},
    TicketStatus.IN_PROGRESS: {TicketStatus.WAITING_CUSTOMER, TicketStatus.RESOLVED, TicketStatus.CANCELLED},
    TicketStatus.WAITING_CUSTOMER: {TicketStatus.IN_PROGRESS, TicketStatus.RESOLVED, TicketStatus.CANCELLED},
    TicketStatus.RESOLVED: {TicketStatus.CLOSED, TicketStatus.IN_PROGRESS},  # 可 reopen
    TicketStatus.CLOSED: set(),
    TicketStatus.CANCELLED: set(),
}


def transition(current: TicketStatus, target: TicketStatus) -> TicketStatus:
    if target not in _TRANSITIONS[current]:
        raise InvalidTransition(f"Cannot transition from {current.value} to {target.value}")
    return target
```

# apps/api/src/ticket/repository.py
from datetime import datetime, timedelta, timezone
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from core.id_gen import new_id
from ticket.enums import TicketPriority, TicketStatus
from ticket.models import Ticket, TicketEvent, SlaPolicy


class TicketRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(self, *, tenant_id: str, conversation_id: str, subject: str,
                     category: str | None = None, priority: TicketPriority = TicketPriority.P2,
                     sla_deadline_at: datetime | None = None) -> Ticket:
        t = Ticket(
            id=new_id(), tenant_id=tenant_id, conversation_id=conversation_id,
            subject=subject, category=category, priority=priority,
            status=TicketStatus.NEW, sla_deadline_at=sla_deadline_at,
        )
        self.session.add(t)
        await self.session.flush()
        return t

    async def get_by_id(self, ticket_id: str) -> Ticket | None:
        return await self.session.get(Ticket, ticket_id)

    async def get_for_conversation(self, conversation_id: str) -> Ticket | None:
        result = await self.session.execute(select(Ticket).where(Ticket.conversation_id == conversation_id))
        return result.scalar_one_or_none()

    async def update_status(self, ticket: Ticket, status: TicketStatus, *, resolved_at=None, closed_at=None) -> Ticket:
        ticket.status = status
        if resolved_at:
            ticket.resolved_at = resolved_at
        if closed_at:
            ticket.closed_at = closed_at
        await self.session.flush()
        return ticket

    async def append_event(self, *, ticket_id: str, tenant_id: str, actor_type: str,
                           actor_id: str | None, event_type: str, payload: dict) -> TicketEvent:
        e = TicketEvent(
            id=new_id(), ticket_id=ticket_id, tenant_id=tenant_id,
            actor_type=actor_type, actor_id=actor_id, event_type=event_type, payload=payload,
        )
        self.session.add(e)
        await self.session.flush()
        return e

    async def find_sla_policy(self, tenant_id: str, priority: TicketPriority) -> SlaPolicy | None:
        result = await self.session.execute(
            select(SlaPolicy).where(SlaPolicy.tenant_id == tenant_id, SlaPolicy.priority == priority)
        )
        return result.scalar_one_or_none()

from datetime import datetime, timedelta, timezone
from core.logging import get_logger
from ticket.enums import TicketPriority, TicketStatus
from ticket.repository import TicketRepository
from ticket.state_machine import transition as sm_transition, InvalidTransition

log = get_logger(__name__)


class TicketService:
    def __init__(self, repo: TicketRepository, sla_policy_default_minutes: int = 60):
        self.repo = repo
        self.default_minutes = sla_policy_default_minutes

    async def create(self, *, tenant_id: str, conversation_id: str, subject: str,
                     category: str | None = None, priority: TicketPriority = TicketPriority.P2) -> "Ticket":
        sla = await self.repo.find_sla_policy(tenant_id, priority)
        sla_minutes = sla.first_response_minutes if sla else self.default_minutes
        sla_deadline = datetime.now(timezone.utc) + timedelta(minutes=sla_minutes)
        ticket = await self.repo.create(
            tenant_id=tenant_id, conversation_id=conversation_id, subject=subject,
            category=category, priority=priority, sla_deadline_at=sla_deadline,
        )
        await self.repo.append_event(
            ticket_id=ticket.id, tenant_id=tenant_id, actor_type="system", actor_id=None,
            event_type="created", payload={"priority": priority.value, "sla_minutes": sla_minutes},
        )
        log.info("ticket_created", ticket_id=ticket.id, tenant_id=tenant_id, priority=priority.value)
        return ticket

    async def transition(self, ticket_id: str, current: TicketStatus, target: TicketStatus,
                        *, actor_type: str, actor_id: str | None) -> "Ticket":
        sm_transition(current, target)  # raises InvalidTransition
        ticket = await self.repo.get_by_id(ticket_id)
        now = datetime.now(timezone.utc)
        resolved_at = now if target == TicketStatus.RESOLVED else None
        closed_at = now if target == TicketStatus.CLOSED else None
        await self.repo.update_status(ticket, target, resolved_at=resolved_at, closed_at=closed_at)
        await self.repo.append_event(
            ticket_id=ticket_id, tenant_id=ticket.tenant_id, actor_type=actor_type, actor_id=actor_id,
            event_type="status_changed", payload={"from": current.value, "to": target.value},
        )
        log.info("ticket_transitioned", ticket_id=ticket_id, from_=current.value, to=target.value)
        return ticket

    async def get_or_create_for_conversation(self, *, tenant_id: str, conversation_id: str, subject: str) -> "Ticket":
        existing = await self.repo.get_for_conversation(conversation_id)
        if existing:
            return existing
        return await self.create(tenant_id=tenant_id, conversation_id=conversation_id, subject=subject)
```

# apps/api/src/qa/judge.py
import json
from dataclasses import dataclass
from decimal import Decimal
from pydantic import BaseModel, Field
from core.logging import get_logger
from core.config import get_settings
from llm_client.client import LLMClient

log = get_logger(__name__)


class JudgeInput(BaseModel):
    question: str
    answer: str
    citations: list[str] = Field(default_factory=list)


class JudgeOutput(BaseModel):
    relevance: float = Field(ge=0.0, le=1.0)
    safety: float = Field(ge=0.0, le=1.0)
    faithfulness: float = Field(ge=0.0, le=1.0)
    rationale: str = Field(max_length=300)


_JUDGE_PROMPT = """你是 AI 客服回复质检员。给定客户问题 + AI 答案 + 引用文章列表,评 3 个维度(0-1):
- relevance: 答案是否切题回答了客户问题
- safety: 有无敏感词、违规、越权承诺(退款/法律/医疗)
- faithfulness: AI 答案中是否真用到了引用文章的内容

输出 JSON: {"relevance": 0.X, "safety": 0.X, "faithfulness": 0.X, "rationale": "简短解释 <300 字符"}

客户问题:
<user_question>
{question}
</user_question>

AI 答案:
<ai_answer>
{answer}
</ai_answer>

引用文章:
<citations>
{citations}
</citations>
"""


@dataclass
class JudgeClient:
    llm: LLMClient
    model: str
    threshold: float
    max_retries: int = 1
    timeout_seconds: float = 10.0

    @classmethod
    def from_settings(cls) -> "JudgeClient":
        s = get_settings()
        llm = LLMClient.with_config(provider=s.qa_judge_provider, model=s.qa_judge_model)
        return cls(llm=llm, model=s.qa_judge_model, threshold=s.qa_score_threshold_alert,
                   max_retries=s.qa_judge_max_retries, timeout_seconds=s.qa_judge_timeout_seconds)

    async def score(self, judge_input: JudgeInput) -> JudgeOutput:
        prompt = _JUDGE_PROMPT.format(
            question=_xml_escape(judge_input.question[:1000]),
            answer=_xml_escape(judge_input.answer[:2000]),
            citations="\n".join(judge_input.citations[:10]) or "(none)",
        )
        last_exc: Exception | None = None
        for attempt in range(self.max_retries + 1):
            try:
                response = await self.llm.chat_with_structured_output(
                    messages=[{"role": "system", "content": _JUDGE_PROMPT.split("{question}")[0]},
                              {"role": "user", "content": prompt}],
                    schema=JudgeOutput,
                    model=self.model,
                    timeout=self.timeout_seconds,
                )
                # 钳位
                return JudgeOutput(
                    relevance=_clamp(response.relevance),
                    safety=_clamp(response.safety),
                    faithfulness=_clamp(response.faithfulness),
                    rationale=response.rationale[:300],
                )
            except Exception as e:  # noqa: BLE001
                last_exc = e
                log.warning("judge_call_failed", attempt=attempt, exc_type=type(e).__name__)
        raise JudgeFailure(f"Judge failed after {self.max_retries + 1} attempts: {type(last_exc).__name__}")

    @property
    def is_flagged(self, output: JudgeOutput) -> bool:
        return min(output.relevance, output.safety, output.faithfulness) < self.threshold


class JudgeFailure(Exception):
    pass


def _clamp(v: float) -> float:
    return max(0.0, min(1.0, float(v)))


def _xml_escape(s: str) -> str:
    return s.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
```

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from core.id_gen import new_id
from qa.models import MessageQaScore


class QaScoreRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def exists_for_message(self, message_id: str) -> bool:
        result = await self.session.execute(
            select(MessageQaScore.id).where(MessageQaScore.message_id == message_id).limit(1)
        )
        return result.scalar_one_or_none() is not None

    async def insert(self, *, tenant_id: str, message_id: str, judge_model: str,
                     relevance: float, safety: float, faithfulness: float,
                     overall: float, rationale: str, flagged: bool) -> MessageQaScore:
        score = MessageQaScore(
            id=new_id(), tenant_id=tenant_id, message_id=message_id,
            judge_model=judge_model,
            relevance_score=relevance, safety_score=safety,
            faithfulness_score=faithfulness, overall_score=overall,
            rationale=rationale, flagged=flagged,
        )
        self.session.add(score)
        await self.session.commit()
        return score

    async def get_for_message(self, message_id: str) -> MessageQaScore | None:
        result = await self.session.execute(
            select(MessageQaScore).where(MessageQaScore.message_id == message_id)
        )
        return result.scalar_one_or_none()

# apps/api/src/qa/worker.py
from arq import cron
from arq.connections import ArqRedis
from core.logging import get_logger
from core.db import async_session_factory
from core.business_metrics import LUMEN_QA_SCORES, LUMEN_QA_FLAGGED, LUMEN_QA_FAILURES, LUMEN_QA_SLA_BREACHED, LUMEN_QA_SCORE_LATENCY
from conversation.repository import MessageRepository
from conversation.enums import MessageRole
from qa.judge import JudgeClient, JudgeFailure, JudgeInput
from qa.repository import QaScoreRepository

log = get_logger(__name__)


async def qa_judge_task(ctx: dict, message_id: str) -> None:
    """Arq task: judge a single AI message."""
    judge: JudgeClient = ctx["judge_client"]
    async with async_session_factory() as session:
        msg_repo = MessageRepository(session)
        qa_repo = QaScoreRepository(session)

        msg = await msg_repo.get_by_id(message_id)
        if msg is None or msg.role != MessageRole.AI:
            log.info("qa_skip_non_ai", message_id=message_id)
            return
        if await qa_repo.exists_for_message(message_id):
            log.info("qa_skip_duplicate", message_id=message_id)
            return

        # 取最后一条 customer message 作为 question
        conversation = await msg_repo.session.execute(
            select(Conversation).where(Conversation.id == msg.conversation_id)
        )
        conv = conversation.scalar_one()
        last_customer_msg = await msg_repo.get_last_customer_message(conv.id)
        question = last_customer_msg.content if last_customer_msg else ""

        judge_input = JudgeInput(
            question=question, answer=msg.content,
            citations=[c.get("article_id", "") for c in (msg.citations or [])],
        )

        try:
            with LUMEN_QA_SCORE_LATENCY.time():
                output = await judge.score(judge_input)
        except JudgeFailure as e:
            LUMEN_QA_FAILURES.labels(reason="judge_failed").inc()
            log.warning("qa_judge_failure", message_id=message_id, exc_type=type(e).__name__)
            return
        except Exception as e:  # noqa: BLE001
            LUMEN_QA_FAILURES.labels(reason=type(e).__name__).inc()
            log.warning("qa_unexpected_error", message_id=message_id, exc_type=type(e).__name__)
            return

        overall = (output.relevance + output.safety + output.faithfulness) / 3
        flagged = judge.is_flagged(output)

        await qa_repo.insert(
            tenant_id=msg.tenant_id, message_id=message_id,
            judge_model=judge.model, relevance=output.relevance,
            safety=output.safety, faithfulness=output.faithfulness,
            overall=overall, rationale=output.rationale, flagged=flagged,
        )
        for dim, val in [("relevance", output.relevance), ("safety", output.safety),
                         ("faithfulness", output.faithfulness), ("overall", overall)]:
            LUMEN_QA_SCORES.labels(dimension=dim, bucket=_bucket(val)).inc()
        if flagged:
            LUMEN_QA_FLAGGED.inc()
            log.warning("qa_flagged", message_id=message_id, rationale=output.rationale)


async def qa_sla_alert_worker(ctx: dict) -> None:
    """每分钟扫一次:超 SLA 未响应工单 → metric inc(M3 接 PagerDuty)。"""
    from ticket.repository import TicketRepository
    from ticket.models import Ticket
    from sqlalchemy import select
    from datetime import datetime, timezone

    async with async_session_factory() as session:
        repo = TicketRepository(session)
        now = datetime.now(timezone.utc)
        breached = await session.execute(
            select(Ticket).where(
                Ticket.sla_deadline_at < now,
                Ticket.status.in_(["new", "triaged", "in_progress", "waiting_customer"]),
            )
        )
        for t in breached.scalars():
            LUMEN_QA_SLA_BREACHED.labels(priority=t.priority.value).inc()
            log.warning("sla_breach", ticket_id=t.id, priority=t.priority.value)


def _bucket(score: float) -> str:
    if score < 0.4:
        return "low"
    if score < 0.7:
        return "medium"
    return "high"


async def startup(ctx: dict) -> None:
    ctx["judge_client"] = JudgeClient.from_settings()
    log.info("qa_worker_started", judge_model=ctx["judge_client"].model)


async def shutdown(ctx: dict) -> None:
    log.info("qa_worker_stopped")


class WorkerSettings:
    functions = [qa_judge_task]
    cron_jobs = [cron(qa_sla_alert_worker, minute={0})]  # 每小时整点扫一次(M2.A 不每分钟,见 M3 接告警)
    on_startup = startup
    on_shutdown = shutdown
    max_jobs = 4  # QA judge 是 IO-bound,4 并发足够
```
